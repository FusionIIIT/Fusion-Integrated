from __future__ import annotations

from typing import Optional
from rest_framework.exceptions import PermissionDenied, NotFound
from modules.lms.models import Course, CourseMembership


def get_course_membership(course_id: int, user_id: int) -> Optional[CourseMembership]:
    """Retrieve active course membership for user."""
    return CourseMembership.objects.filter(
        course_id=course_id, user_id=user_id, is_active=True
    ).first()


def require_course_access(course_id: int, user_id: int) -> tuple[Course, CourseMembership]:
    """Verify user is enrolled, assigned, or linked to the course (BR-LMS-001).

    Returns the course and membership. Raises NotFound / PermissionDenied if unlinked.
    """
    course = Course.objects.filter(id=course_id, is_active=True).first()
    if not course:
        raise NotFound("Course not found.")

    membership = get_course_membership(course_id, user_id)
    if not membership:
        raise PermissionDenied("You do not have access to this course workspace.")

    return course, membership


def check_faculty_or_ta(membership: CourseMembership) -> bool:
    return membership.role in ("faculty", "ta")


def require_faculty(membership: CourseMembership, error_msg="Only course faculty may perform this action."):
    if membership.role != "faculty":
        raise PermissionDenied(error_msg)


def require_can_announce(course: Course, membership: CourseMembership):
    """BR-LMS-007: Faculty may post announcements; TA may post only where permitted."""
    if membership.role == "faculty":
        return
    if membership.role == "ta" and course.allow_ta_announcements:
        return
    raise PermissionDenied("You do not have permission to post announcements in this course.")


def require_can_grade(course: Course, membership: CourseMembership):
    """BR-LMS-015, BR-LMS-004: Faculty or TA (if enabled) can grade submissions."""
    if membership.role == "faculty":
        return
    if membership.role == "ta" and course.allow_ta_grading:
        return
    raise PermissionDenied("You do not have grading permissions in this course.")


def require_non_observer(membership: CourseMembership, error_msg: str = "Observers have read-only access and cannot access student-specific data."):
    """BR-LMS-003: Observers cannot view student submissions, grades, or perform writes."""
    if membership.role == "observer":
        raise PermissionDenied(error_msg)

"""Grading and grade release services (UC-LMS-012, UC-LMS-013, BR-LMS-015, BR-LMS-019)."""
from __future__ import annotations

from decimal import Decimal
from django.utils import timezone
from rest_framework.exceptions import ValidationError, PermissionDenied
from modules.lms.models import (
    Assignment,
    AssignmentSubmission,
    SubmissionGrade,
    GradeChangeAudit,
    CourseMembership,
)


def grade_submission(
    submission: AssignmentSubmission,
    points: Decimal | None = None,
    feedback: str = "",
    grader_user_id: int = 0,
    grader_role: str = "",
    grader_name: str = "",
    change_reason: str = "",
    points_awarded: Decimal | None = None,
) -> SubmissionGrade:
    """Record or update a submission grade with attribution (BR-LMS-015).

    If grade was already released, record an audit entry (BR-LMS-019).
    """
    if points is None and points_awarded is not None:
        points = points_awarded
    if points is None:
        raise ValidationError("Points awarded is required.")

    if points < 0:
        raise ValidationError("Points awarded cannot be negative.")
    if points > submission.assignment.points_possible:
        raise ValidationError(
            f"Points awarded ({points}) exceeds points possible ({submission.assignment.points_possible})."
        )

    grade = getattr(submission, "grade", None)
    if grade is None:
        # Create initial grade in draft state (BR-LMS-016)
        grade = SubmissionGrade.objects.create(
            submission=submission,
            points_awarded=points,
            feedback=feedback,
            status="draft",
            graded_by_user_id=grader_user_id,
            graded_by_role=grader_role,
            graded_by_name=grader_name,
        )
    else:
        # If grade was already released, create audit log before updating (BR-LMS-019)
        if grade.status == "released":
            GradeChangeAudit.objects.create(
                grade=grade,
                previous_points=grade.points_awarded,
                new_points=points,
                previous_feedback=grade.feedback,
                new_feedback=feedback,
                changed_by_user_id=grader_user_id,
                changed_by_name=grader_name,
                change_reason=change_reason or "Grade revised post-release",
            )

        grade.points_awarded = points
        grade.feedback = feedback
        grade.graded_by_user_id = grader_user_id
        grade.graded_by_role = grader_role
        grade.graded_by_name = grader_name
        grade.save()

    return grade


def release_assignment_grades(assignment: Assignment, faculty_user_id: int) -> int:
    """Release all draft grades for an assignment to students (BR-LMS-017)."""
    now = timezone.now()
    updated = SubmissionGrade.objects.filter(
        submission__assignment=assignment,
        status="draft",
    ).update(
        status="released",
        released_at=now,
        released_by_user_id=faculty_user_id,
    )
    return updated


def release_single_grade(grade: SubmissionGrade, faculty_user_id: int) -> SubmissionGrade:
    """Release a single student's draft grade."""
    grade.status = "released"
    grade.released_at = timezone.now()
    grade.released_by_user_id = faculty_user_id
    grade.save()
    return grade

class CourseListView(APIView):

    @extend_schema(operation_id="lms_courses_list", responses=CourseSerializer(many=True))

        actor = _actor(request)
        user_id = actor.user_id if actor else None

        # BR-LMS-001, BR-LMS-002: Return courses the user is enrolled in or assigned to
        memberships = CourseMembership.objects.filter(user_id=user_id, is_active=True).values_list("course_id", "role")
        user_course_ids = [m[0] for m in memberships]
        role_map = {m[0]: m[1] for m in memberships}

        courses = Course.objects.filter(id__in=user_course_ids, is_active=True).annotate(
            materials_count=Count("materials", filter=Q(materials__status="published"), distinct=True),

class AssignmentSubmitView(APIView):
    permission_classes = [MODULE]

    @extend_schema(operation_id="lms_assignment_submit", request=AssignmentSubmissionSerializer, responses=AssignmentSubmissionSerializer)
    def post(self, request, assignment_id: int):
        actor = _actor(request)
        asgn = Assignment.objects.filter(id=assignment_id).first()
        if not asgn:
            raise NotFound("Assignment not found.")
        _, membership = require_course_access(asgn.course_id, actor.user_id)
        require_non_observer(membership, "Observers cannot submit assignments (BR-LMS-003).")

        if membership.role != "student":
            raise PermissionDenied("Only enrolled students may submit assignment solutions.")

        now = timezone.now()
        is_late = now > asgn.due_date
        if is_late and not asgn.allow_late_submissions:
            raise ValidationError("Late submissions are not accepted for this assignment (BR-LMS-012).")

        previous_sub = AssignmentSubmission.objects.filter(
            assignment=asgn, student_user_id=actor.user_id
        ).order_by("-version").first()
        version = (previous_sub.version + 1) if previous_sub else 1

        serializer = AssignmentSubmissionSerializer(data=request.data, context={"request": request})
        serializer.is_valid(raise_exception=True)

        sub_type = serializer.validated_data.get("submission_type")
        if sub_type and sub_type != asgn.submission_type:
            raise ValidationError(f"Submission type '{sub_type}' does not match required type '{asgn.submission_type}' (BR-LMS-012).")

        submission = serializer.save(
            assignment=asgn,
            student_user_id=actor.user_id,
            status="late" if is_late else "submitted",
            submitted_at=now,
            version=version,
        )
        return Response(AssignmentSubmissionSerializer(submission, context={"request": request}).data, status=status.HTTP_201_CREATED)
