  ```python
  @transaction.atomic
  def post(postings: Iterable[Posting], *, created_by_user_id: int | None = None) -> list[LeaveLedgerEntry]:
      written: list[LeaveLedgerEntry] = []
      for posting in postings:
          if posting.days < 0:
              raise LedgerError("Post a negative quantity as a debit kind; day amounts are magnitudes.")
          balance = _locked_balance(posting.user_id, posting.year, posting.category)
          balance.days = (balance.days + posting.days if posting.kind in ADDITIVE
                          else balance.days - posting.days)
          entry = LeaveLedgerEntry.objects.create(
              user_id=posting.user_id, year=posting.year, category=posting.category,
              kind=posting.kind, days=posting.days, note=posting.note,
              leave_request_id=posting.leave_request_id,
              created_by_user_id=created_by_user_id)
          balance.save(update_fields=["days", "updated_at"])
          written.append(entry)
      return written

  @transaction.atomic
  def reverse(entry: LeaveLedgerEntry, *, note: str = "", created_by_user_id: int | None = None) -> LeaveLedgerEntry:
      if entry.kind == REVERSAL:
          raise LedgerError("A reversal cannot itself be reversed; post a new entry.")
      mirror = LeaveLedgerEntry.objects.filter(reverses_id=entry.pk).first()
      if mirror is not None:
          raise LedgerError("This entry has already been reversed.")
      if entry.days == 0:
          raise LedgerError("A zero entry has nothing to reverse.")
      balance = _locked_balance(entry.user_id, entry.year, entry.category)
      if entry.kind not in ADDITIVE and balance.days < entry.days:
          raise LedgerError(
              f"Reversing {entry.days} {entry.category} would take the balance below zero (currently {balance.days}).")
      balance.days = (balance.days - entry.days if entry.kind in ADDITIVE
                      else balance.days + entry.days)
      balance.save(update_fields=["days", "updated_at"])
      return LeaveLedgerEntry.objects.create(
          user_id=entry.user_id, year=entry.year, category=entry.category,
          kind=REVERSAL, days=entry.days, note=note or f"Reverses #{entry.pk}",
          leave_request_id=entry.leave_request_id, reverses_id=entry.pk,
          created_by_user_id=created_by_user_id)
  ```

```python
from decimal import Decimal

from modules.leave.models import LeaveBalance, LeaveLedgerEntry
from modules.leave.services.ledger import post, reverse

def process_year_end(user_id: int, year: int):
    # Lapse CL, RH, and SCL balances
    for category in ["CL", "RH", "SCL"]:
        balance = LeaveBalance.objects.get(user_id=user_id, year=year, category=category)
        balance.days = Decimal("0")
        balance.save()

        # Record the lapse in the ledger
        posting = Posting(user_id=user_id, year=year, category=category, kind="DEBIT", days=Decimal("0"))
        post([posting], created_by_user_id=user_id)

    # Carry forward EL and COL balances
    for category in ["EL", "COL"]:
        prev_year_balance = LeaveBalance.objects.get(user_id=user_id, year=year-1, category=category)
        balance = LeaveBalance.objects.get(user_id=user_id, year=year, category=category)
        balance.days = prev_year_balance.days
        balance.save()

        # Record the carry forward in the ledger
        posting = Posting(user_id=user_id, year=year, category=category, kind="CREDIT", days=prev_year_balance.days)
        post([posting], created_by_user_id=user_id)

    # Convert faculty VL to EL at 2:1 ratio
    vl_balance = LeaveBalance.objects.get(user_id=user_id, year=year-1, category="VL")
    if vl_balance.days > 0:
        el_balance = LeaveBalance.objects.get(user_id=user_id, year=year, category="EL")
        el_balance.days += vl_balance.days / Decimal("2")
        el_balance.save()

        # Record the conversion in the ledger
        posting = Posting(user_id=user_id, year=year, category="EL", kind="CREDIT", days=vl_balance.days / Decimal("2"))
```

   ```
   """Year-end balance outcomes (CFR-EL-028, BR-EL-002..009).

   Pure arithmetic over balances the caller supplies, so the year-end sweep is a
   thin transactional wrapper and the figures are testable without a database.
   """
   from __future__ import annotations

   from dataclasses import dataclass
   from decimal import ROUND_HALF_UP, Decimal

   from modules.leave.domain import categories as cat
   from modules.leave.domain.policy import Policy


   @dataclass(frozen=True)
   class Outcome:
       """One ledger movement the year-end run must post."""
       category: str
       kind: str
       days: Decimal
       note: str


   @dataclass(frozen=True)
   class Balance:
       user_id: int
       category: str
       days: Decimal


   def _convert_vacation(unused: Decimal, ratio: Decimal) -> Decimal:
       """BR-EL-007: 2 VL : 1 EL, fractions included."""
       if unused <= 0 or ratio <= 0:
           return Decimal("0")
       return (unused / ratio).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)


   def plan(policy: Policy, kind: str, balances: list[Balance],
            year: int) -> list[Outcome]:
       """What year end does to one employee's balances.

       CL, RH and SCL lapse (BR-EL-002/003/004). Faculty VL converts to EL and
       never carries forward as VL (BR-EL-007/008). EL and COL carry forward
       (BR-EL-005/009), subject to whatever cap the policy carries.
       """
       out: list[Outcome] = []
       for balance in balances:
           unused = balance.days
           if unused <= 0:
               continue
           category = balance.category

           if category == cat.VACATION:
               if kind != cat.FACULTY:
                   out.append(Outcome(category, "lapse", unused,
                                      "Vacation Leave cannot be carried forward."))
                   continue
               out.append(Outcome(category, "lapse", unused,
                                  "Unused Vacation Leave does not carry forward."))
               converted = _convert_vacation(unused, policy.vacation_to_earned_ratio)
               if converted > 0:
                   out.append(Outcome(cat.EARNED, "conversion", converted,
                                      "Vacation Leave converted to Earned Leave."))
               continue

           if category not in cat.CARRIES_FORWARD:
               out.append(Outcome(category, "lapse", unused,
                                  "Unused leave of this category lapses at year end."))
               continue

           if not policy.carries_forward(category):
               out.append(Outcome(category, "lapse", unused,
                                  "Carry-forward is switched off for this category."))
               continue

           cap = policy.carry_forward_cap(category)
           carried = unused if cap is None else min(unused, cap)
           out.append(Outcome(category, "carry_forward", carried,
                              "Unused balance carried into the next year."))
           if carried < unused:
               out.append(Outcome(category, "lapse", unused - carried,
                                  "Unused balance above the carry-forward cap lapses."))
       return out
   ```

   ```
   """BR-EL-002..009: the year-end run.

   Runs once a year, credits the new entitlements, lapses what the policy does not
   carry forward, converts Vacation Leave into Earned Leave at the configured
   ratio, and carries the balance over. Everything is driven by `domain.year_end`,
   so the arithmetic is testable without a database and the run only supplies rows.
   """
   from __future__ import annotations

   import logging
   from dataclasses import dataclass, field
   from decimal import Decimal

   from django.conf import settings
   from django.db import transaction
   from django.utils import timezone

   from modules.directory import contracts as directory
   from modules.leave.domain import year_end as year_end_domain
   from modules.leave.models import (
       CARRY_FORWARD,
       CONVERSION,
       CREDIT,
       LAPSE,
       LeaveBalance,
       LeavePolicy,
   )
   from modules.leave.services import ledger, notifications

   log = logging.getLogger("fusion.leave.yearend")


   @dataclass
   class YearEndReport:
       year: int
       credited: int = 0
       lapsed: int = 0
       converted: int = 0
       carried: int = 0
       skipped: int = 0
       exceptions: list[dict] = field(default_factory=list)

       def __str__(self) -> str:
           return (f"{self.year}: credited={self.credited} lapsed={self.lapsed} "
                   f"converted={self.converted} carried={self.carried} "
                   f"skipped={self.skipped} exceptions={len(self.exceptions)}")


   def is_due(*, today, policy: LeavePolicy) -> bool:
       """BR-EL-001: the run belongs to the closing year. Run early only when an
       administrator says so, never on a calendar guess alone."""
       return today.month >= policy.year_end_month


   def people_in_scope() -> list[int]:
       """Everyone on the payroll. The directory owns that definition."""
       try:
           return sorted(person.user_id for person in directory.get_employees())
       except Exception as exc:                                       # noqa: BLE001
           log.warning("leave.yearend.directory_unavailable err=%s", exc)
           return []


   @transaction.atomic
   def run(*, year: int, user_ids: list[int] | None = None, now=None,
           dry_run: bool = False) -> YearEndReport:
       """Close `year` and open `year + 1`. Idempotent per person: a person whose
       balances are already marked processed is skipped rather than credited twice
       (BR-EL-002)."""
       now = now or timezone.now()
       pol = LeavePolicy.objects.first() or LeavePolicy()
       pol_domain = pol.as_domain()
       report = YearEndReport(year=year)

       for user_id in (user_ids if user_ids is not None else people_in_scope()):
           try:
               _run_for_person(user_id, year=year, pol_domain=pol_domain, now=now,
                               dry_run=dry_run, report=report)
           except Exception as exc:                                   # noqa: BLE001
               # One person's arithmetic never stops the run for the rest (BR-EL-009).
               log.exception("leave.yearend.person_failed user=%s year=%s", user_id, year)
               report.exceptions.append({"user_id": user_id, "error": str(exc)})
               notifications.enqueue(
                   topic="leave.year_end.exception",
                   dedupe_key=f"year-end-exception:{year}:{user_id}",
                   recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
                   subject=f"Leave year-end failed for user {user_id}",
                   body=f"Year {year}: {exc}")

       if not dry_run:
           _credit_entitlements(pol, pol_domain, report)
       notifications.enqueue(
           topic="leave.year_end.processed",
           dedupe_key=f"year-end:{year}",
           recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
           subject=f"Leave year-end {year} processed",
           body=str(report))
       log.info("leave.yearend.done %s", report)
       return report


   def _run_for_person(user_id: int, *, year: int, pol_domain, now, dry_run: bool,
                       report: YearEndReport) -> None:
       opening = (LeaveBalance.objects
                  .filter(user_id=user_id, year=year + 1)
                  .exists())
       if opening:
           # Already run for this person: doing it twice would double-credit them.
           report.skipped += 1
           return

       balances = list(LeaveBalance.objects.filter(user_id=user_id, year=year))
       plan = year_end_domain.plan(balances, pol_domain, year)

       postings = []
       for decision in plan:
           if decision.action == year_end_domain.LAPSE:
               postings.append(ledger.Posting(
                   user_id=user_id, year=year, category=decision.category, kind=LAPSE,
                   days=decision.days, note=f"Year-end lapse of {year}"))
               report.lapsed += 1
           elif decision.action == year_end_domain.CARRY:
               postings.append(ledger.Posting(
                   user_id=user_id, year=year + 1, category=decision.category,
                   kind=CARRY_FORWARD, days=decision.days,
                   note=f"Carried forward from {year}"))
               report.carried += 1
           elif decision.action == year_end_domain.CONVERT:
               postings.append(ledger.Posting(
                   user_id=user_id, year=year + 1, category=decision.category,
                   kind=CONVERSION, days=decision.days,
                   note=f"Vacation converted to Earned Leave for {year + 1}"))
               report.converted += 1

       if dry_run:
           return
       if postings:
           ledger.post(postings, created_by_user_id=None)
       LeaveBalance.objects.filter(user_id=user_id, year=year + 1).update(
           year_end_applied=True)


   def _credit_entitlements(pol: LeavePolicy, pol_domain, report: YearEndReport) -> None:
       """BR-EL-004..008: the new year's entitlement, per person kind. Credit is the
       policy's configured figure, not a hard-coded one."""
       next_year = report.year + 1
       for person in directory.get_employees():
           entitlement = _entitlement_for(person.kind, pol)
           if entitlement is None:
               continue
           category = "VL" if person.kind == "faculty" else "EL"
           ledger.post([ledger.Posting(
               user_id=person.user_id, year=next_year, category=category, kind=CREDIT,
               days=Decimal(str(entitlement)),
               note=f"Entitlement for {next_year}")])
           report.credited += 1


   def _entitlement_for(kind: str, pol: LeavePolicy) -> float | None:
       return {"faculty": float(pol.vacation_days),
               "staff": float(pol.earned_days)}.get(kind)
   ```

   ```
           return sorted(person.user_id for person in directory.get_employees())
       except Exception as exc:                                       # noqa: BLE001
           log.warning("leave.yearend.directory_unavailable err=%s", exc)
           return []


   @transaction.atomic
   def run(*, year: int, user_ids: list[int] | None = None, now=None,
           dry_run: bool = False) -> YearEndReport:
       """Close `year` and open `year + 1`. Idempotent per person: a person whose
       balances are already marked processed is skipped rather than credited twice
       (BR-EL-002)."""
       now = now or timezone.now()
       pol = LeavePolicy.objects.first() or LeavePolicy()
       pol_domain = pol.as_domain()
       report = YearEndReport(year=year)

       for user_id in (user_ids if user_ids is not None else people_in_scope()):
           try:
               _run_for_person(user_id, year=year, pol_domain=pol_domain, now=now,
                               dry_run=dry_run, report=report)
           except Exception as exc:                                   # noqa: BLE001
               # One person's arithmetic never stops the run for the rest (BR-EL-009).
               log.exception("leave.yearend.person_failed user=%s year=%s", user_id, year)
               report.exceptions.append({"user_id": user_id, "error": str(exc)})
               notifications.enqueue(
                   topic="leave.year_end.exception",
                   dedupe_key=f"year-end-exception:{year}:{user_id}",
                   recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
                   subject=f"Leave year-end failed for user {user_id}",
                   body=f"Year {year}: {exc}")

       if not dry_run:
           _credit_entitlements(pol, pol_domain, report)
       notifications.enqueue(
           topic="leave.year_end.processed",
           dedupe_key=f"year-end:{year}",
           recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
           subject=f"Leave year-end {year} processed",
           body=str(report))
       log.info("leave.yearend.done %s", report)
       return report


   def _run_for_person(user_id: int, *, year: int, pol_domain, now, dry_run: bool,
                       report: YearEndReport) -> None:
       opening = (LeaveBalance.objects
                  .filter(user_id=user_id, year=year + 1)
                  .exists())
       if opening:
           # Already run for this person: doing it twice would double-credit them.
           report.skipped += 1
           return

       balances = list(LeaveBalance.objects.filter(user_id=user_id, year=year))
       plan = year_end_domain.plan(balances, pol_domain, year)

       postings = []
       for decision in plan:
           if decision.action == year_end_domain.LAPSE:
               postings.append(ledger.Posting(
                   user_id=user_id, year=year, category=decision.category, kind=LAPSE,
                   days=decision.days, note=f"Year-end lapse of {year}"))
               report.lapsed += 1
           elif decision.action == year_end_domain.CARRY:
               postings.append(ledger.Posting(
                   user_id=user_id, year=year + 1, category=decision.category,
                   kind=CARRY_FORWARD, days=decision.days,
                   note=f"Carried forward from {year}"))
               report.carried += 1
           elif decision.action == year_end_domain.CONVERT:
               postings.append(ledger.Posting(
                   user_id=user_id, year=year + 1, category=decision.category,
                   kind=CONVERSION, days=decision.days,
                   note=f"Vacation converted to Earned Leave for {year + 1}"))
               report.converted += 1

       if dry_run:
           return
       if postings:
           ledger.post(postings, created_by_user_id=None)
       LeaveBalance.objects.filter(user_id=user_id, year=year + 1).update(
           year_end_applied=True)


   def _credit_entitlements(pol: LeavePolicy, pol_domain, report: YearEndReport) -> None:
       """BR-EL-004..008: the new year's entitlement, per person kind. Credit is the
       policy's configured figure, not a hard-coded one."""
       next_year = report.year + 1
       for person in directory.get_employees():
           entitlement = _entitlement_for(person.kind, pol)
           if entitlement is None:
               continue
           category = "VL" if person.kind == "faculty" else "EL"
           ledger.post([ledger.Posting(
               user_id=person.user_id, year=next_year, category=category, kind=CREDIT,
               days=Decimal(str(entitlement)),
               note=f"Entitlement for {next_year}")])
           report.credited += 1


   def _entitlement_for(kind: str, pol: LeavePolicy) -> float | None:
       return {"faculty": float(pol.vacation_days),
               "staff": float(pol.earned_days)}.get(kind)
   ```

```
"""BR-EL-002..009: the year-end run.

Runs once a year, credits the new entitlements, lapses what the policy does not
carry forward, converts Vacation Leave into Earned Leave at the configured
ratio, and carries the balance over. Everything is driven by `domain.year_end`,
so the arithmetic is testable without a database and the run only supplies rows.

from dataclasses import dataclass, field
from decimal import Decimal
```

```
@dataclass
class YearEndReport:
    year: int
    credited: int = 0
    lapsed: int = 0
    converted: int = 0
    carried: int = 0
    skipped: int = 0
    exceptions: list[dict] = field(default_factory=list)

    def __str__(self) -> str:
        return (f"{self.year}: credited={self.credited} lapsed={self.lapsed} "
                f"converted={self.converted} carried={self.carried} "
                f"skipped={self.skipped} exceptions={len(self.exceptions)}")


def is_due(*, today, policy: LeavePolicy) -> bool:
    """BR-EL-001: the run belongs to the closing year. Run early only when an
    administrator says so, never on a calendar guess alone."""
    return today.month >= policy.year_end_month


def people_in_scope() -> list[int]:
    """Everyone on the payroll. The directory owns that definition."""
    try:
        return sorted(person.user_id for person in directory.get_employees())
    except Exception as exc:                                       # noqa: BLE001
        log.warning("leave.yearend.directory_unavailable err=%s", exc)
        return []


@transaction.atomic
def run(*, year: int, user_ids: list[int] | None = None, now=None,
        dry_run: bool = False) -> YearEndReport:
    """Close `year` and open `year + 1`. Idempotent per person: a person whose
    balances are already marked processed is skipped rather than credited twice
    (BR-EL-002)."""
    now = now or timezone.now()
    pol = LeavePolicy.objects.first() or LeavePolicy()
    pol_domain = pol.as_domain()
    report = YearEndReport(year=year)

    # Create a mapping of user_id to kind
    employee_kind_map = {user.user_id: user.kind for user in directory.get_employees()}

    for user_id in (user_ids if user_ids is not None else people_in_scope()):
        try:
            employee_kind = employee_kind_map[user_id]  # Get employee kind from the mapping
            _run_for_person(user_id, year=year, pol_domain=pol_domain, now=now,
                            dry_run=dry_run, report=report)
        except Exception as exc:                                   # noqa: BLE001
            # One person's arithmetic never stops the run for the rest (BR-EL-009).
            log.exception("leave.yearend.person_failed user=%s year=%s", user_id, year)
            report.exceptions.append({"user_id": user_id, "error": str(exc)})
            notifications.enqueue(
                topic="leave.year_end.exception",
                dedupe_key=f"year-end-exception:{year}:{user_id}",
                recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
                subject=f"Leave year-end failed for user {user_id}",
                body=f"Year {year}: {exc}")

    if not dry_run:
        _credit_entitlements(pol, pol_domain, report)
    notifications.enqueue(
        topic="leave.year_end.processed",
        dedupe_key=f"year-end:{year}",
        recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
        subject=f"Leave year-end {year} processed",
        body=str(report))
    log.info("leave.yearend.done %s", report)
    return report


def _run_for_person(user_id: int, *, year: int, pol_domain, now, dry_run: bool,
                    report: YearEndReport) -> None:
    opening = (LeaveBalance.objects
               .filter(user_id=user_id, year=year + 1)
               .exists())
    if opening:
        # Already run for this person: doing it twice would double-credit them.
        report.skipped += 1
        return

    balances = list(LeaveBalance.objects.filter(user_id=user_id, year=year))
    employee_kind = employee_kind_map[user_id]  # Get employee kind from the mapping
    plan = year_end_domain.plan(pol_domain, employee_kind, balances, year)

    postings = []
    for decision in plan:
        if decision.action == year_end_domain.LAPSE:
            postings.append(ledger.Posting(
                user_id=user_id, year=year, category=decision.category, kind=LAPSE,
                days=decision.days, note=f"Year-end lapse of {year}"))
            report.lapsed += 1
        elif decision.action == year_end_domain.CARRY:
            postings.append(ledger.Posting(
                user_id=user_id, year=year + 1, category=decision.category,
                kind=CARRY_FORWARD, days=decision.days,
                note=f"Carried forward from {year}"))
            report.carried += 1
        elif decision.action == year_end_domain.CONVERT:
            postings.append(ledger.Posting(
                user_id=user_id, year=year + 1, category=decision.category,
                kind=CONVERSION, days=decision.days,
                note=f"Vacation converted to Earned Leave for {year + 1}"))
            report.converted += 1

    if dry_run:
        return
    if postings:
        ledger.post(postings, created_by_user_id=None)
    LeaveBalance.objects.filter(user_id=user_id, year=year + 1).update(
        year_end_applied=True)


def _credit_entitlements(pol: LeavePolicy, pol_domain, report: YearEndReport) -> None:
    """BR-EL-004..008: the new year's entitlement, per person kind. Credit is the
    policy's configured figure, not a hard-coded one."""
    next_year = report.year + 1
    for person in directory.get_employees():
        entitlement = _entitlement_for(person.kind, pol)
        if entitlement is None:
            continue
        category = "VL" if person.kind == "faculty" else "EL"
        ledger.post([ledger.Posting(
            user_id=person.user_id, year=next_year, category=category, kind=CREDIT,
            days=Decimal(str(entitlement)),
            note=f"Entitlement for {next_year}")])
        report.credited += 1


def _entitlement_for(kind: str, pol: LeavePolicy) -> float | None:
    return {"faculty": float(pol.vacation_days),
            "staff": float(pol.earned_days)}.get(kind)
```

```python
@dataclass
class YearEndReport:
    year: int
    credited: int = 0
    lapsed: int = 0
    converted: int = 0
    carried: int = 0
    skipped: int = 0
    exceptions: list[dict] = field(default_factory=list)

        return (f"{self.year}: credited={self.credited} lapsed={self.lapsed} "
                f"converted={self.converted} carried={self.carried} "
                f"skipped={self.skipped} exceptions={len(self.exceptions)}")

def is_due(*, today, policy: LeavePolicy) -> bool:
    return today.month >= policy.year_end_month

def people_in_scope() -> list[int]:
    try:
        return sorted(person.user_id for person in directory.get_employees())
    except Exception as exc:
        log.warning("leave.yearend.directory_unavailable err=%s", exc)
        return []

@transaction.atomic
def run(*, year: int, user_ids: list[int] | None = None, now=None, dry_run: bool = False) -> YearEndReport:
    now = now or timezone.now()
    pol = LeavePolicy.objects.first() or LeavePolicy()
    pol_domain = pol.as_domain()
    report = YearEndReport(year=year)

    employee_kind_map = {user.user_id: user.kind for user in directory.get_employees()}

    for user_id in (user_ids if user_ids is not None else people_in_scope()):
        try:
            employee_kind = employee_kind_map[user_id]
            _run_for_person(
                user_id, year=year, pol_domain=pol_domain, now=now,
                dry_run=dry_run, report=report, employee_kind=employee_kind,
            )
        except Exception as exc:
            log.exception("leave.yearend.person_failed user=%s year=%s", user_id, year)
            report.exceptions.append({"user_id": user_id, "error": str(exc)})
            notifications.enqueue(
                topic="leave.year_end.exception",
                dedupe_key=f"year-end-exception:{year}:{user_id}",
                recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
                subject=f"Leave year-end failed for user {user_id}",
                body=f"Year {year}: {exc}")

    if not dry_run:
        _credit_entitlements(pol, pol_domain, report)
    notifications.enqueue(
        topic="leave.year_end.processed",
        dedupe_key=f"year-end:{year}",
        recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
        subject=f"Leave year-end {year} processed",
        body=str(report))
    log.info("leave.yearend.done %s", report)
    return report

def _run_for_person(user_id: int, *, year: int, pol_domain, now, dry_run: bool, report: YearEndReport, employee_kind: str) -> None:
    opening = (LeaveBalance.objects
               .filter(user_id=user_id, year=year + 1)
               .exists())
    if opening:
        report.skipped += 1
        return

    balances = list(LeaveBalance.objects.filter(user_id=user_id, year=year))
    plan = year_end_domain.plan(pol_domain, employee_kind, balances, year)
    postings = []
    for decision in plan:
        if decision.action == year_end_domain.LAPSE:
            postings.append(ledger.Posting(
                user_id=user_id, year=year, category=decision.category, kind=LAPSE,
                days=decision.days, note=f"Year-end lapse of {year}"))
            report.lapsed += 1
        elif decision.action == year_end_domain.CARRY:
            postings.append(ledger.Posting(
                user_id=user_id, year=year + 1, category=decision.category,
                kind=CARRY_FORWARD, days=decision.days,
                note=f"Carried forward from {year}"))
            report.carried += 1
        elif decision.action == year_end_domain.CONVERT:
            postings.append(ledger.Posting(
                user_id=user_id, year=year + 1, category=decision.category,
                kind=CONVERSION, days=decision.days,
                note=f"Vacation converted to Earned Leave for {year + 1}"))
            report.converted += 1

    if dry_run:
        return
    if postings:
        ledger.post(postings, created_by_user_id=None)
    LeaveBalance.objects.filter(user_id=user_id, year=year + 1).update(
        year_end_applied=True)

def _credit_entitlements(pol: LeavePolicy, pol_domain, report: YearEndReport) -> None:
    next_year = report.year + 1
    for person in directory.get_employees():
        entitlement = _entitlement_for(person.kind, pol)
        if entitlement is None:
            continue
        category = "VL" if person.kind == "faculty" else "EL"
        ledger.post([ledger.Posting(
            user_id=person.user_id, year=next_year, category=category, kind=CREDIT,
            days=Decimal(str(entitlement)),
            note=f"Entitlement for {next_year}")])
        report.credited += 1

def _entitlement_for(kind: str, pol: LeavePolicy) -> float | None:
    return {"faculty": float(pol.vacation_days),
            "staff": float(pol.earned_days)}.get(kind)
```

```
Runs once a year, credits the new entitlements, lapses what the policy does not
carry forward, converts Vacation Leave into Earned Leave at the configured

    converted: int = 0

def is_due(*, today, policy: LeavePolicy) -> bool:
    """BR-EL-001: the run belongs to the closing year. Run early only when an
    administrator says so, never on a calendar guess alone."""
    return today.month >= policy.year_end_month


def people_in_scope() -> list[int]:
    """Everyone on the payroll. The directory owns that definition."""
    try:
        return sorted(person.user_id for person in directory.get_employees())
    except Exception as exc:  # noqa: BLE001 - directory failure must not stop callers
        log.warning("leave.yearend.directory_unavailable err=%s", exc)
        return []


@transaction.atomic
def run(*, year: int, user_ids: list[int] | None = None, now=None,
        dry_run: bool = False) -> YearEndReport:
    """Close `year` and open `year + 1`. Idempotent per person: a person whose
    balances are already marked processed is skipped rather than credited twice
    (BR-EL-002)."""
    now = now or timezone.now()
    pol = LeavePolicy.objects.first() or LeavePolicy()
    pol_domain = pol.as_domain()
    report = YearEndReport(year=year)

    # Create a mapping of user_id to kind
    employee_kind_map = {user.user_id: user.kind for user in directory.get_employees()}

    for user_id in (user_ids if user_ids is not None else people_in_scope()):
        try:
            employee_kind = employee_kind_map[user_id]
            _run_for_person(
                user_id, year=year, pol_domain=pol_domain, now=now,
                dry_run=dry_run, report=report, employee_kind=employee_kind,
            )
        except Exception as exc:
            # One person's arithmetic never stops the run for the rest (BR-EL-009).
            log.exception("leave.yearend.person_failed user=%s year=%s", user_id, year)
            report.exceptions.append({"user_id": user_id, "error": str(exc)})
            notifications.enqueue(
                topic="leave.year_end.exception",
                dedupe_key=f"year-end-exception:{year}:{user_id}",
                recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
                subject=f"Leave year-end failed for user {user_id}",
                body=f"Year {year}: {exc}")

    if not dry_run:
        _credit_entitlements(pol, pol_domain, report)
    notifications.enqueue(
        topic="leave.year_end.processed",
        dedupe_key=f"year-end:{year}",
        recipient_email=getattr(settings, "OPS_EMAIL", "ops@invalid"),
        subject=f"Leave year-end {year} processed",
        body=str(report))
    log.info("leave.yearend.done %s", report)
    return report


def _run_for_person(user_id: int, *, year: int, pol_domain, now, dry_run: bool,
                    report: YearEndReport, employee_kind: str) -> None:
    opening = (LeaveBalance.objects
               .filter(user_id=user_id, year=year + 1)
               .exists())
    if opening:
        # Already run for this person: doing it twice would double-credit them.
        report.skipped += 1
        return

    balances = list(LeaveBalance.objects.filter(user_id=user_id, year=year))
    plan = year_end_domain.plan(pol_domain, employee_kind, balances, year)
    postings = []
    for decision in plan:
        if decision.kind == year_end_domain.LAPSE:
            postings.append(ledger.Posting(
                user_id=user_id, year=year, category=decision.category, kind=LAPSE,
                days=decision.days, note=f"Year-end lapse of {year}"))
            report.lapsed += 1
        elif decision.kind == year_end_domain.CARRY:
            postings.append(ledger.Posting(
                user_id=user_id, year=year + 1, category=decision.category,
                kind=CARRY_FORWARD, days=decision.days,
                note=f"Carried forward from {year}"))
            report.carried += 1
        elif decision.kind == year_end_domain.CONVERT:
            postings.append(ledger.Posting(
                user_id=user_id, year=year + 1, category=decision.category,
                kind=CONVERSION, days=decision.days,
                note=f"Vacation converted to Earned Leave for {year + 1}"))
            report.converted += 1

    if dry_run:
        return
    if postings:
        ledger.post(postings, created_by_user_id=None)
    LeaveBalance.objects.filter(user_id=user_id, year=year + 1).update(
        year_end_applied=True)


def _credit_entitlements(pol: LeavePolicy, pol_domain, report: YearEndReport) -> None:
    """BR-EL-004..008: the new year's entitlement, per person kind. Credit is the
    policy's configured figure, not a hard-coded one."""
    next_year = report.year + 1
    for person in directory.get_employees():
        entitlement = _entitlement_for(person.kind, pol)
        if entitlement is None:
            continue
        category = "VL" if person.kind == "faculty" else "EL"
        ledger.post([ledger.Posting(
            user_id=person.user_id, year=next_year, category=category, kind=CREDIT,
            days=Decimal(str(entitlement)),
            note=f"Entitlement for {next_year}")])
        report.credited += 1


def _entitlement_for(kind: str, pol: LeavePolicy) -> float | None:
    return {"faculty": float(pol.vacation_days),
            "staff": float(pol.earned_days)}.get(kind)
```
