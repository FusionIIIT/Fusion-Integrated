      });

    e.preventDefault();
    if (!selectedType || !startDate || !endDate) {

import React, { useState } from 'react';
import { Paper, Title, Select, TextInput, Button, Stack } from '@mantine/core';
import { useAuth } from 'client/src/modules/auth/hooks/useAuth';
import { http } from 'client/src/lib/http';

const LeaveRequestForm: React.FC<{ onSuccess?: () => void }> = ({ onSuccess }) => {
    const [form] = useMantineForm({
        initialValues: {
            leaveType: '',
            startDate: '',
            endDate: '',
        },
    });

    const { user } = useAuth();
    const [leaveTypes, setLeaveTypes] = useState([]);

    useEffect(() => {
        http.get('/leave/types/')
            .then(res => {
                setLeaveTypes(Array.isArray(res.data) ? res.data : res.data.results);
            })
            .catch(err => {
                console.error('Failed to fetch leave types:', err);
            });
    }, []);

    const onSubmit = (values) => {
        const { leaveType, startDate, endDate } = values;
        http.post('/leave/applications/', {
            leave_type: Number(leaveType),
            start_date: startDate,
            end_date: endDate,
            status: 'pending',
        })
        .then(() => {
            if (onSuccess) onSuccess();
        })
        .catch(err => {
            console.error('Failed to submit leave application:', err);
        });
    };

    return (
        <Paper p="md">
            <Title order={3}>Leave Request Form</Title>
            <Stack>
                <Select
                    label="Leave Type"
                    data={leaveTypes.map(type => ({ value: type.id, label: type.name }))}
                    {...form.getInputProps('leaveType')}
                />
                <TextInput
                    label="Start Date"
                    type="date"
                    {...form.getInputProps('startDate')}
                />
                <TextInput
                    label="End Date"
                    type="date"
                    {...form.getInputProps('endDate')}
                />
                <Button type="submit" onClick={() => form.onSubmit(onSubmit)}>
                    Submit
                </Button>
            </Stack>
        </Paper>
    );
};

export default LeaveRequestForm;
