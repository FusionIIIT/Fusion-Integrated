from rest_framework.views import APIView
from rest_framework.response import Response
from fusion_auth.permissions import IsAuthenticatedPrincipal, HasModuleGrant, HasPermission
from drf_spectacular.utils import extend_schema
import json

class ConfigurationView(APIView):
    permission_classes = [IsAuthenticatedPrincipal, HasModuleGrant("leave_elm"), HasPermission("leave_elm.configuration.manage")]

    @extend_schema(responses=OpenApiTypes.OBJECT)
    def get(self, request):
        try:
            policy = LeavePolicy.objects.get(user=request.principal.user_id)
        except LeavePolicy.DoesNotExist:
            policy = LeavePolicy(name='Default', user_id=request.principal.user_id)
            policy.save()

        try:
            calendar = LeaveCalendar.objects.get(user_id=request.principal.user_id)
        except LeaveCalendar.DoesNotExist:
            calendar = LeaveCalendar(name='Default', user_id=request.principal.user_id, holidays=[])
            calendar.save()

        policies = LeavePolicySerializer(LeavePolicy.objects.all(), many=True).data
        calendars = LeaveCalendarSerializer(LeaveCalendar.objects.all(), many=True).data

        return Response({'policy': LeavePolicySerializer(policy).data, 'calendar': LeaveCalendarSerializer(calendar).data, 'policies': policies, 'calendars': calendars})

    def post(self, request):
        if request.method == 'POST':
            data = json.loads(request.body.decode('utf-8'))
            policy_serializer = LeavePolicySerializer(data=data['policy'], partial=True)
            calendar_serializer = LeaveCalendarSerializer(data=data['calendar'])

            if policy_serializer.is_valid() and calendar_serializer.is_valid():
                policy_serializer.save(user=request.principal.user_id)
                calendar_serializer.save(user=request.principal.user_id)

                response_data = {
                    'policy': policy_serializer.data,
                    'calendar': calendar_serializer.data
                }

                if 'run_rollover' in data and data['run_rollover']:
                    # Assuming year_end_rollover is a method on LeavePolicy or LeaveCalendar model
                    policy.year_end_rollover()
                    calendar.year_end_rollover()

                return Response(response_data, status=201)
            else:
                return Response(policy_serializer.errors or calendar_serializer.errors, status=400)
