@override_settings(FUSION_AUTH_CLIENT_ID="test", FUSION_AUTH_SECRET_KEY="test")
def _make_principal(
    user_id: int, *, kind='faculty', designations=None, permissions=None
) -> Principal:
    if not designations:
        designations = ['Faculty']
    
    if not permissions:
        permissions = ROLE_GRANTS['faculty']

    session_data = {
        'user_id': user_id,
        'username': f'user_{user_id}',
        'display_name': f'User {user_id}',
        'kind': kind,
        'active_role': None,
        'roles': tuple(designations),
        'permissions': frozenset(permissions),
        'modules': ('leave_elm',),
        'email': '',
        'navigation': (),
    }
    return Principal.from_session(session_data)

        'CL': LEAVE_POLICY_DEFAULTS['CL'],

@transaction.atomic
def test_cancelation(faculty_principal):
    app = submit_application(
        actor=faculty_principal,
        data={
            'leave_type': 'CL',
            'start_date': date(2023, 1, 1),
            'end_date': date(2023, 1, 5),
            'purpose': 'Test Leave Application',
            'holidays': (),
            'restricted_holidays': (),
            'prefix_suffix_days': 0,
            'academic_substitute_user_id': None,
            'admin_substitute_user_id': None,
            'attachment_reference': '',
            'remarks': ''
        }
    )
    
    act_on_application(faculty_principal, app.id, 'cancel')

    app.refresh_from_db()
    assert app.status == 'Cancelled'
    assert app.cancelled_by_id == faculty_principal.user_id

    balances = get_leave_balances([faculty_principal.user_id])
    expected_balance = LEAVE_POLICY_DEFAULTS['CL'] - Decimal('4.00')
    for balance in balances:
        cat_name = balance.leave_category.name
        if cat_name == 'CL':
            assert balance.balance == Decimal(str(expected_balance))
        else:
            assert balance.balance == Decimal(str(LEAVE_POLICY_DEFAULTS[cat_name]))
